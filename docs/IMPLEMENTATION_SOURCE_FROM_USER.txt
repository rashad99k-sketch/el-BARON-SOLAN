"""Single managed PumpPortal WebSocket connection.

Docs: https://pumpportal.fun/data-api/real-time/
- subscribeNewToken, subscribeMigration are free, no key.
- Docs forbid one connection per token/wallet. One connection, both subs.
- Provider-side timestamp is stored separately from receive timestamp.
"""
from __future__ import annotations
import asyncio
import json
import logging
import random
import time
from dataclasses import dataclass, field
from typing import Awaitable, Callable, Optional

import websockets

log = logging.getLogger("grok6.providers.pumpportal")

WS_URI = "wss://pumpportal.fun/api/data"


@dataclass
class PumpPortalState:
    connected: bool = False
    last_message_at: Optional[float] = None
    last_provider_event_at: Optional[float] = None
    reconnect_count: int = 0
    last_error: Optional[str] = None
    messages_received: int = 0
    new_tokens_seen: int = 0
    migrations_seen: int = 0
    duplicates_suppressed: int = 0
    schema_errors: int = 0
    started_at: Optional[float] = None


class PumpPortalProvider:
    name = "pumpportal"

    def __init__(
        self,
        on_new_token: Callable[[dict], Awaitable[None]],
        on_migration: Callable[[dict], Awaitable[None]],
        max_backoff_seconds: float = 60.0,
        max_consecutive_failures: int = 20,
    ):
        self.on_new_token = on_new_token
        self.on_migration = on_migration
        self.max_backoff_seconds = max_backoff_seconds
        self.max_consecutive_failures = max_consecutive_failures
        self.state = PumpPortalState()
        self._stop = asyncio.Event()
        self._seen_new: set[str] = set()
        self._seen_migrations: set[str] = set()

    async def run(self) -> None:
        """Main loop. Capped exponential backoff + jitter. Bounded consecutive failures."""
        self.state.started_at = time.time()
        backoff = 1.0
        failures = 0
        while not self._stop.is_set():
            try:
                async with websockets.connect(
                    WS_URI, ping_interval=20, ping_timeout=20, close_timeout=5
                ) as ws:
                    self.state.connected = True
                    self.state.last_error = None
                    failures = 0
                    backoff = 1.0
                    log.info("pumpportal connected")

                    await ws.send(json.dumps({"method": "subscribeNewToken"}))
                    await ws.send(json.dumps({"method": "subscribeMigration"}))

                    async for raw in ws:
                        self.state.last_message_at = time.time()
                        self.state.messages_received += 1
                        await self._handle(raw)

            except asyncio.CancelledError:
                break
            except Exception as e:
                self.state.connected = False
                self.state.last_error = f"{type(e).__name__}: {e}"
                self.state.reconnect_count += 1
                failures += 1
                log.warning("pumpportal disconnect: %s", self.state.last_error)
                if failures >= self.max_consecutive_failures:
                    log.error("pumpportal exceeded max failures; stopping")
                    break
                jitter = random.uniform(0, backoff * 0.3)
                await asyncio.sleep(min(backoff, self.max_backoff_seconds) + jitter)
                backoff = min(backoff * 2, self.max_backoff_seconds)

        self.state.connected = False

    async def stop(self) -> None:
        self._stop.set()

    async def _handle(self, raw: str) -> None:
        try:
            msg = json.loads(raw)
        except json.JSONDecodeError:
            self.state.schema_errors += 1
            return
        if not isinstance(msg, dict):
            self.state.schema_errors += 1
            return

        mint = msg.get("mint") or msg.get("tokenAddress")
        if not isinstance(mint, str) or not mint:
            self.state.schema_errors += 1
            return

        provider_ts = msg.get("timestamp")
        if isinstance(provider_ts, (int, float)):
            ts = float(provider_ts)
            if ts > 1e12:
                ts /= 1000.0
            self.state.last_provider_event_at = ts

        is_migration = bool(msg.get("migration") or msg.get("pool"))

        if is_migration:
            if mint in self._seen_migrations:
                self.state.duplicates_suppressed += 1
                return
            self._seen_migrations.add(mint)
            self.state.migrations_seen += 1
            await self.on_migration(msg)
        else:
            if mint in self._seen_new:
                self.state.duplicates_suppressed += 1
                return
            self._seen_new.add(mint)
            self.state.new_tokens_seen += 1
            await self.on_new_token(msg)

    def health(self) -> dict:
        now = time.time()
        age = None if self.state.last_message_at is None else now - self.state.last_message_at
        if self.state.started_at is None:
            status = "NOT_CONFIGURED"
        elif self.state.connected and (age is None or age < 90):
            status = "CONNECTED"
        elif self.state.connected:
            status = "DEGRADED"
        else:
            status = "DISCONNECTED"
        return {
            "provider": self.name,
            "status": status,
            "connected": self.state.connected,
            "last_message_age_seconds": age,
            "last_provider_event_age_seconds":
                None if self.state.last_provider_event_at is None
                else now - self.state.last_provider_event_at,
            "reconnect_count": self.state.reconnect_count,
            "messages_received": self.state.messages_received,
            "new_tokens_seen": self.state.new_tokens_seen,
            "migrations_seen": self.state.migrations_seen,
            "duplicates_suppressed": self.state.duplicates_suppressed,
            "schema_errors": self.state.schema_errors,
            "last_error": self.state.last_error,
        }

"""Solana JSON-RPC adapter: mint account data, SPL Token + Token-2022 parsing.

- Reads program owner before interpreting bytes.
- Distinguishes DISABLED / PRESENT / UNKNOWN / ERROR for each authority.
- Never estimates top-holder concentration from incomplete data.
"""
from __future__ import annotations
import base64
import logging
import time
from dataclasses import dataclass
from enum import Enum
from typing import Optional

import requests

log = logging.getLogger("grok6.providers.solana_rpc")

SPL_TOKEN_PROGRAM = "TokenkegQfeZyiNwAJbNbGKPFXCWuBvf9Ss623VQ5DA"
TOKEN_2022_PROGRAM = "TokenzQdBNbLqP5VEhdkAS6EPFLC1PHnBqCXEpPxuEb"


class AuthorityStatus(str, Enum):
    DISABLED = "DISABLED"
    PRESENT = "PRESENT"
    UNKNOWN = "UNKNOWN"
    ERROR = "ERROR"


@dataclass
class MintAuthorityInfo:
    mint: str
    owner_program: Optional[str]
    mint_authority: AuthorityStatus
    freeze_authority: AuthorityStatus
    decimals: Optional[int]
    supply_raw: Optional[int]
    slot: Optional[int]
    source: str = "solana_rpc"
    fetched_at: float = 0.0
    error: Optional[str] = None


def _parse_spl_mint(raw: bytes) -> tuple[AuthorityStatus, AuthorityStatus, int, int]:
    """SPL Token Mint layout:
       0..4    mint_authority_option (u32 LE)
       4..36   mint_authority (Pubkey)
       36..44  supply (u64 LE)
       44..45  decimals (u8)
       45..46  is_initialized (u8)
       46..50  freeze_authority_option (u32 LE)
    Token-2022 reuses the same base 82 bytes; extensions follow.
    """
    if len(raw) < 82:
        raise ValueError(f"mint data too short: {len(raw)}")
    mint_opt = int.from_bytes(raw[0:4], "little")
    supply = int.from_bytes(raw[36:44], "little")
    decimals = raw[44]
    freeze_opt = int.from_bytes(raw[46:50], "little")
    mint_a = AuthorityStatus.DISABLED if mint_opt == 0 else AuthorityStatus.PRESENT
    freeze_a = AuthorityStatus.DISABLED if freeze_opt == 0 else AuthorityStatus.PRESENT
    return mint_a, freeze_a, decimals, supply


class SolanaRpcProvider:
    name = "solana_rpc"

    def __init__(self, rpc_url: str, timeout: float = 8.0, max_retries: int = 3):
        self.rpc_url = rpc_url
        self.timeout = timeout
        self.max_retries = max_retries
        self.session = requests.Session()
        self._last_success_at: Optional[float] = None
        self._last_error: Optional[str] = None
        self._requests_made = 0
        self._errors = 0

    def _rpc(self, method: str, params: list) -> Optional[dict]:
        body = {"jsonrpc": "2.0", "id": 1, "method": method, "params": params}
        backoff = 0.4
        for _ in range(self.max_retries):
            self._requests_made += 1
            try:
                r = self.session.post(self.rpc_url, json=body, timeout=self.timeout)
                if r.status_code == 429:
                    time.sleep(backoff); backoff *= 2; continue
                r.raise_for_status()
                return r.json().get("result")
            except Exception as e:
                self._last_error = f"{type(e).__name__}: {e}"
                self._errors += 1
                time.sleep(backoff); backoff *= 2
        return None

    def fetch_mint_authorities(self, mint: str) -> MintAuthorityInfo:
        now = time.time()
        res = self._rpc("getAccountInfo", [mint, {"encoding": "base64"}])
        if not res or not res.get("value"):
            return MintAuthorityInfo(mint, None, AuthorityStatus.UNKNOWN,
                                     AuthorityStatus.UNKNOWN, None, None, None,
                                     fetched_at=now, error="no_account")
        acct = res["value"]
        owner = acct.get("owner", "")
        slot = res.get("context", {}).get("slot")
        data = acct.get("data")
        if not (isinstance(data, list) and len(data) >= 1):
            return MintAuthorityInfo(mint, owner, AuthorityStatus.UNKNOWN,
                                     AuthorityStatus.UNKNOWN, None, None, slot,
                                     fetched_at=now, error="bad_data_shape")
        try:
            raw = base64.b64decode(data[0])
        except Exception as e:
            return MintAuthorityInfo(mint, owner, AuthorityStatus.ERROR,
                                     AuthorityStatus.ERROR, None, None, slot,
                                     fetched_at=now, error=f"b64:{e}")

        if owner not in (SPL_TOKEN_PROGRAM, TOKEN_2022_PROGRAM):
            return MintAuthorityInfo(mint, owner, AuthorityStatus.UNKNOWN,
                                     AuthorityStatus.UNKNOWN, None, None, slot,
                                     fetched_at=now,
                                     error=f"unknown_owner_program:{owner}")

        try:
            mint_a, freeze_a, decimals, supply = _parse_spl_mint(raw)
        except Exception as e:
            return MintAuthorityInfo(mint, owner, AuthorityStatus.ERROR,
                                     AuthorityStatus.ERROR, None, None, slot,
                                     fetched_at=now, error=f"parse:{e}")

        self._last_success_at = now
        return MintAuthorityInfo(mint, owner, mint_a, freeze_a, decimals, supply,
                                 slot, fetched_at=now)

    def health(self) -> dict:
        now = time.time()
        age = None if self._last_success_at is None else now - self._last_success_at
        if self._requests_made == 0:
            status = "NOT_CONFIGURED"
        elif self._last_error and (age is None or age > 600):
            status = "DEGRADED"
        else:
            status = "CONNECTED"
        return {
            "provider": self.name,
            "status": status,
            "rpc_url": self.rpc_url,
            "requests_made": self._requests_made,
            "errors": self._errors,
            "last_success_age_seconds": age,
            "last_error": self._last_error,
        }

"""DexScreener enrichment. Never treated as discovery source.

Docs: https://docs.dexscreener.com/api/reference
- No universal "new pairs" endpoint. pairCreatedAt may be missing.
- Missing pairCreatedAt -> UNKNOWN. Never inferred as fresh.
"""
from __future__ import annotations
import logging
import time
from dataclasses import dataclass
from typing import Optional

import requests

log = logging.getLogger("grok6.providers.dexscreener")

BASE = "https://api.dexscreener.com"


@dataclass
class DexScreenerEnrichment:
    mint: str
    status: str                       # OK | NO_PAIR_YET | ERROR
    pool_id: Optional[str] = None
    symbol: Optional[str] = None
    price_usd: Optional[float] = None
    liquidity_usd: Optional[float] = None
    volume_5m_usd: Optional[float] = None
    volume_1h_usd: Optional[float] = None
    price_change_5m_pct: Optional[float] = None
    price_change_1h_pct: Optional[float] = None
    buys_5m: Optional[int] = None
    sells_5m: Optional[int] = None
    pair_created_at_ms: Optional[int] = None
    source: str = "dexscreener"
    fetched_at: float = 0.0
    error: Optional[str] = None


class DexScreenerProvider:
    name = "dexscreener"

    def __init__(self, timeout: float = 8.0, cache_ttl: float = 15.0):
        self.timeout = timeout
        self.cache_ttl = cache_ttl
        self.session = requests.Session()
        self._cache: dict[str, tuple[DexScreenerEnrichment, float]] = {}
        self._last_success_at: Optional[float] = None
        self._last_error: Optional[str] = None
        self._errors = 0

    def enrich(self, mint: str) -> DexScreenerEnrichment:
        cached = self._cache.get(mint)
        if cached and time.time() - cached[1] < self.cache_ttl:
            return cached[0]
        now = time.time()
        try:
            r = self.session.get(f"{BASE}/latest/dex/tokens/{mint}", timeout=self.timeout)
            r.raise_for_status()
            data = r.json()
        except Exception as e:
            self._last_error = f"{type(e).__name__}: {e}"
            self._errors += 1
            ev = DexScreenerEnrichment(mint, "ERROR", fetched_at=now,
                                       error=self._last_error)
            self._cache[mint] = (ev, now)
            return ev

        pairs = [p for p in (data.get("pairs") or []) if p.get("chainId") == "solana"]
        if not pairs:
            ev = DexScreenerEnrichment(mint, "NO_PAIR_YET", fetched_at=now)
            self._cache[mint] = (ev, now)
            return ev

        p = max(pairs, key=lambda x: (x.get("liquidity") or {}).get("usd", 0) or 0)
        liq = p.get("liquidity") or {}
        vol = p.get("volume") or {}
        chg = p.get("priceChange") or {}
        txns = (p.get("txns") or {}).get("m5") or {}

        def _f(x):
            try:
                return float(x) if x is not None else None
            except (TypeError, ValueError):
                return None

        created = p.get("pairCreatedAt")
        ev = DexScreenerEnrichment(
            mint=mint, status="OK",
            pool_id=p.get("pairAddress"),
            symbol=(p.get("baseToken") or {}).get("symbol"),
            price_usd=_f(p.get("priceUsd")),
            liquidity_usd=_f(liq.get("usd")),
            volume_5m_usd=_f(vol.get("m5")),
            volume_1h_usd=_f(vol.get("h1")),
            price_change_5m_pct=_f(chg.get("m5")),
            price_change_1h_pct=_f(chg.get("h1")),
            buys_5m=txns.get("buys"),
            sells_5m=txns.get("sells"),
            pair_created_at_ms=int(created) if isinstance(created, (int, float)) else None,
            fetched_at=now,
        )
        self._last_success_at = now
        self._cache[mint] = (ev, now)
        return ev

    def health(self) -> dict:
        now = time.time()
        age = None if self._last_success_at is None else now - self._last_success_at
        if self._last_success_at is None and self._errors == 0:
            status = "NOT_CONFIGURED"
        elif self._errors and (age is None or age > 300):
            status = "DEGRADED"
        else:
            status = "CONNECTED"
        return {
            "provider": self.name,
            "status": status,
            "last_success_age_seconds": age,
            "errors": self._errors,
            "last_error": self._last_error,
        }

"""RugCheck API v1 client. https://api.rugcheck.xyz/v1

CRITICAL: `rugged` is NOT `honeypot`. Keep them separate claims.
Missing data is not a pass. API failure is UNKNOWN / DATA_SOURCE_ERROR.

Rate limits (verified 2026-10-11):
- without key: 10 req/min
- with API key: 60 req/min
"""
from __future__ import annotations
import logging
import time
from dataclasses import dataclass, field
from typing import Optional

import requests

log = logging.getLogger("grok6.providers.rugcheck")

BASE = "https://api.rugcheck.xyz/v1"


@dataclass
class RugCheckEvidence:
    mint: str
    status: str                       # OK | NOT_FOUND | ERROR
    rugged: Optional[bool] = None     # separate claim, never renamed
    score_normalised: Optional[int] = None
    mint_authority_present: Optional[bool] = None
    freeze_authority_present: Optional[bool] = None
    top10_pct: Optional[float] = None
    insider_network_count: Optional[int] = None
    risk_names: list = field(default_factory=list)
    raw_keys: list = field(default_factory=list)
    source: str = "rugcheck"
    fetched_at: float = 0.0
    error: Optional[str] = None


class RugCheckProvider:
    name = "rugcheck"

    def __init__(self, api_key: Optional[str] = None, timeout: float = 8.0,
                 max_retries: int = 2, cache_ttl: float = 120.0,
                 min_interval_seconds: float = 6.5):
        # 10 req/min without key -> min 6s between requests to be safe
        self.api_key = api_key
        self.timeout = timeout
        self.max_retries = max_retries
        self.cache_ttl = cache_ttl
        self.min_interval_seconds = min_interval_seconds if not api_key else 1.1
        self.session = requests.Session()
        self._cache: dict[str, tuple[RugCheckEvidence, float]] = {}
        self._last_request_at: float = 0.0
        self._last_success_at: Optional[float] = None
        self._last_error: Optional[str] = None
        self._requests_made = 0
        self._rate_limited = 0

    def _headers(self) -> dict:
        h = {"Accept": "application/json"}
        if self.api_key:
            h["X-API-KEY"] = self.api_key
        return h

    def enrich(self, mint: str) -> RugCheckEvidence:
        cached = self._cache.get(mint)
        if cached and time.time() - cached[1] < self.cache_ttl:
            return cached[0]

        # rate limit
        elapsed = time.time() - self._last_request_at
        if elapsed < self.min_interval_seconds:
            time.sleep(self.min_interval_seconds - elapsed)
        self._last_request_at = time.time()

        now = time.time()
        url = f"{BASE}/tokens/{mint}/report"
        backoff = 0.5
        for _ in range(self.max_retries):
            self._requests_made += 1
            try:
                r = self.session.get(url, headers=self._headers(), timeout=self.timeout)
                if r.status_code == 429:
                    self._rate_limited += 1
                    time.sleep(backoff); backoff *= 2; continue
                if r.status_code == 404:
                    ev = RugCheckEvidence(mint, "NOT_FOUND", fetched_at=now)
                    self._cache[mint] = (ev, now)
                    return ev
                r.raise_for_status()
                data = r.json()
                ev = self._normalize(mint, data, now)
                self._cache[mint] = (ev, now)
                self._last_success_at = now
                return ev
            except Exception as e:
                self._last_error = f"{type(e).__name__}: {e}"
                time.sleep(backoff); backoff *= 2

        ev = RugCheckEvidence(mint, "ERROR", fetched_at=now,
                              error=self._last_error or "unknown")
        self._cache[mint] = (ev, now)
        return ev

    @staticmethod
    def _normalize(mint: str, d: dict, now: float) -> RugCheckEvidence:
        top = d.get("topHolders") or []
        top10_pct = None
        if isinstance(top, list) and top:
            try:
                top10_pct = sum(float(h.get("pct", 0)) for h in top[:10]) / 100.0
            except (TypeError, ValueError):
                top10_pct = None

        insider = d.get("insiderNetworks")
        insider_count = len(insider) if isinstance(insider, list) else None

        risks = d.get("risks") or []
        names = [r.get("name") for r in risks if isinstance(r, dict) and r.get("name")]

        return RugCheckEvidence(
            mint=mint, status="OK",
            rugged=d.get("rugged"),
            score_normalised=d.get("score_normalised"),
            mint_authority_present=(d.get("mintAuthority") is not None)
                if "mintAuthority" in d else None,
            freeze_authority_present=(d.get("freezeAuthority") is not None)
                if "freezeAuthority" in d else None,
            top10_pct=top10_pct,
            insider_network_count=insider_count,
            risk_names=names,
            raw_keys=sorted(list(d.keys()))[:30],
            fetched_at=now,
        )

    def health(self) -> dict:
        now = time.time()
        age = None if self._last_success_at is None else now - self._last_success_at
        if self._requests_made == 0:
            status = "NOT_CONFIGURED"
        elif self._rate_limited > 0 and (age is None or age > 300):
            status = "QUOTA_EXCEEDED"
        elif self._last_error and (age is None or age > 600):
            status = "DEGRADED"
        else:
            status = "CONNECTED"
        return {
            "provider": self.name,
            "status": status,
            "authenticated": bool(self.api_key),
            "requests_made": self._requests_made,
            "rate_limited_count": self._rate_limited,
            "last_success_age_seconds": age,
            "last_error": self._last_error,
        }

"""Jupiter sell-route feasibility check (read-only).

Docs: https://dev.jup.ag/
- amount is in INPUT token's smallest units, NOT SOL lamports.
- Uses token decimals from verified mint data.

@dataclass
class SellRouteEvidence:
    mint: str
    status: str                       # OK | NO_ROUTE | SELL_ROUTE_UNKNOWN
    amount_sent_smallest_units: Optional[int] = None
    price_impact_pct: Optional[float] = None
    out_amount: Optional[str] = None
    output_mint: str = SOL_MINT
    source: str = "jupiter"
    fetched_at: float = 0.0

class JupiterProvider:
    name = "jupiter"

    def __init__(self, timeout: float = 8.0, cache_ttl: float = 60.0,
                 output_mint: str = SOL_MINT):
        self.timeout = timeout
        self.cache_ttl = cache_ttl
        self.output_mint = output_mint
        self.session = requests.Session()
        self._cache: dict[str, tuple[SellRouteEvidence, float]] = {}
        self._last_success_at: Optional[float] = None
        self._last_error: Optional[str] = None
        self._requests_made = 0

    def check_sell_route(self, mint: str, decimals: Optional[int],
                         reference_price_usd: Optional[float],
                         paper_size_usd: float = 2.0) -> SellRouteEvidence:
        now = time.time()
        if decimals is None:
            return SellRouteEvidence(mint, "SELL_ROUTE_UNKNOWN", fetched_at=now,
                                     error="NO_DECIMALS")
        if reference_price_usd is None or reference_price_usd <= 0:
            return SellRouteEvidence(mint, "SELL_ROUTE_UNKNOWN", fetched_at=now,
                                     error="NO_REFERENCE_PRICE")

        tokens = paper_size_usd / reference_price_usd
        amount = int(tokens * (10 ** decimals))
        if amount <= 0:
            return SellRouteEvidence(mint, "SELL_ROUTE_UNKNOWN", fetched_at=now,
                                     error="AMOUNT_ROUNDS_TO_ZERO")

        key = f"{mint}:{decimals}:{paper_size_usd}:{self.output_mint}"
        cached = self._cache.get(key)
        if cached and now - cached[1] < self.cache_ttl:
            return cached[0]

        params = {
            "inputMint": mint,
            "outputMint": self.output_mint,
            "amount": str(amount),
            "slippageBps": "300",
        }
        self._requests_made += 1
        try:
            r = self.session.get(f"{BASE}/quote", params=params, timeout=self.timeout)
            if r.status_code != 200:
                ev = SellRouteEvidence(mint, "NO_ROUTE", amount_sent_smallest_units=amount,
                                       fetched_at=now, error=f"HTTP_{r.status_code}")
                self._cache[key] = (ev, now)
                return ev
            q = r.json()
            pi = q.get("priceImpactPct")
            ev = SellRouteEvidence(
                mint=mint, status="OK", amount_sent_smallest_units=amount,
                price_impact_pct=float(pi) if pi is not None else None,
                out_amount=q.get("outAmount"),
                fetched_at=now,
            )
            self._cache[key] = (ev, now)
            self._last_success_at = now
            return ev
        except Exception as e:
            self._last_error = f"{type(e).__name__}: {e}"
            ev = SellRouteEvidence(mint, "SELL_ROUTE_UNKNOWN",
                                   amount_sent_smallest_units=amount,
                                   fetched_at=now, error=self._last_error)
            self._cache[key] = (ev, now)
            return ev

    def health(self) -> dict:
        now = time.time()
        age = None if self._last_success_at is None else now - self._last_success_at
        if self._requests_made == 0:
            status = "NOT_CONFIGURED"
        elif self._last_error and (age is None or age > 600):
            status = "DEGRADED"
        else:
            status = "CONNECTED"
        return {
            "provider": self.name,
            "status": status,
            "requests_made": self._requests_made,
            "last_success_age_seconds": age,
            "last_error": self._last_error,
        }

class HeliusProvider:
    name = "helius"

    def __init__(self, api_key: Optional[str], timeout: float = 8.0):
        self.api_key = api_key
        self.timeout = timeout
        self.session = requests.Session()
        self._seen_signatures: set[str] = set()

        return bool(self.api_key)

"""Explicit lifecycle. No candidate sits in WAITING forever."""
from __future__ import annotations
import time
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional


class Stage(str, Enum):
    DISCOVERED = "DISCOVERED"
    NEEDS_MARKET_ENRICHMENT = "NEEDS_MARKET_ENRICHMENT"
    NEEDS_ONCHAIN_CHECK = "NEEDS_ONCHAIN_CHECK"
    NEEDS_RISK_CHECK = "NEEDS_RISK_CHECK"
    READY_FOR_DECISION = "READY_FOR_DECISION"
    ELIGIBLE_PAPER = "ELIGIBLE_PAPER"
    PAPER_OPEN = "PAPER_OPEN"
    # side / terminal
    NEEDS_REFRESH = "NEEDS_REFRESH"
    WAITING_FOR_RETRY = "WAITING_FOR_RETRY"
    STALE_DATA = "STALE_DATA"
    PROVIDER_ERROR = "PROVIDER_ERROR"
    REJECTED = "REJECTED"
    EXPIRED = "EXPIRED"


TERMINAL = {Stage.REJECTED, Stage.EXPIRED, Stage.PAPER_OPEN}


@dataclass
class Transition:
    at: float
    from_stage: str
    to_stage: str
    reason_code: str
    block_reason: str = ""


@dataclass
class Candidate:
    mint: str
    discovery_source: str
    source_event_at: Optional[float] = None
    first_seen_at: float = field(default_factory=time.time)
    stage: Stage = Stage.DISCOVERED
    reason_code: Optional[str] = None
    block_reason: Optional[str] = None
    last_checked_at: Optional[float] = None
    next_retry_at: Optional[float] = None
    expires_at: Optional[float] = None
    retries: int = 0
    max_retries: int = 12
    transitions: list[Transition] = field(default_factory=list)
    symbol: Optional[str] = None
    pool_id: Optional[str] = None

    def transition(self, new_stage: Stage, reason_code: str,
                   block_reason: str = "") -> None:
        if self.stage in TERMINAL and new_stage not in TERMINAL:
            # do not resurrect terminal states silently
            return
        self.transitions.append(Transition(
            at=time.time(), from_stage=self.stage.value, to_stage=new_stage.value,
            reason_code=reason_code, block_reason=block_reason,
        ))
        self.stage = new_stage
        self.reason_code = reason_code
        self.block_reason = block_reason
        self.last_checked_at = time.time()

    def schedule_retry(self, delay_seconds: float,
                       reason_code: str = "RETRY") -> None:
        self.retries += 1
        if self.retries > self.max_retries:
            self.transition(Stage.EXPIRED, "MAX_RETRIES_EXCEEDED")
            return
        self.next_retry_at = time.time() + delay_seconds
        self.transition(Stage.WAITING_FOR_RETRY, f"{reason_code}_{self.retries}")

    def is_expired(self, now: Optional[float] = None) -> bool:
        now = now or time.time()
        return self.expires_at is not None and now >= self.expires_at

    def to_dict(self) -> dict:
        return {
            "mint": self.mint,
            "symbol": self.symbol,
            "pool_id": self.pool_id,
            "discovery_source": self.discovery_source,
            "source_event_at": self.source_event_at,
            "first_seen_at": self.first_seen_at,
            "stage": self.stage.value,
            "reason_code": self.reason_code,
            "block_reason": self.block_reason,
            "last_checked_at": self.last_checked_at,
            "next_retry_at": self.next_retry_at,
            "expires_at": self.expires_at,
            "retries": self.retries,
            "max_retries": self.max_retries,
            "transitions": [t.__dict__ for t in self.transitions],
        }


class CandidateRegistry:
    def __init__(self, default_ttl_seconds: int = 3600):
        self._by_mint: dict[str, Candidate] = {}
        self.default_ttl = default_ttl_seconds

    def get_or_create(self, mint: str, source: str,
                      source_event_at: Optional[float] = None) -> Candidate:
        c = self._by_mint.get(mint)
        if c is None:
            c = Candidate(mint=mint, discovery_source=source,
                          source_event_at=source_event_at,
                          expires_at=time.time() + self.default_ttl)
            self._by_mint[mint] = c
        return c

    def get(self, mint: str) -> Optional[Candidate]:
        return self._by_mint.get(mint)

    def all(self) -> list[Candidate]:
        return list(self._by_mint.values())

    def expire_due(self, now: Optional[float] = None) -> int:
        now = now or time.time()
        n = 0
        for c in self._by_mint.values():
            if c.stage in TERMINAL:
                continue
            if c.is_expired(now):
                c.transition(Stage.EXPIRED, "TTL_EXCEEDED")
                n += 1
        return n

from __future__ import annotations

@dataclass
class ObservedBar:
    bucket_start: float
    bucket_seconds: int
    open: float
    high: float
    low: float
    close: float
    samples: int
    source: str = "observed candles (GROK6-collected)"


class ObservedCandleBuilder:
    def __init__(self, bucket_seconds: int = 60, min_bars_for_chart: int = 3):
        self.bucket_seconds = bucket_seconds
        self.min_bars_for_chart = min_bars_for_chart
        self._bars: dict[str, list[ObservedBar]] = {}

    def add_sample(self, mint: str, price: Optional[float],
                   ts: Optional[float] = None) -> None:
        if price is None or price <= 0:
            return
        ts = ts or time.time()
        bucket = int(ts // self.bucket_seconds) * self.bucket_seconds
        bars = self._bars.setdefault(mint, [])
        if bars and bars[-1].bucket_start == bucket:
            b = bars[-1]
            b.high = max(b.high, price)
            b.low = min(b.low, price)
            b.close = price
            b.samples += 1
        else:
            bars.append(ObservedBar(bucket, self.bucket_seconds,
                                    price, price, price, price, 1))

    def get_bars(self, mint: str) -> list[ObservedBar]:
        return list(self._bars.get(mint, []))

    def coverage(self, mint: str) -> dict:
        bars = self._bars.get(mint, [])
        if len(bars) < self.min_bars_for_chart:
            return {"status": "INSUFFICIENT_HISTORY",
                    "bars": len(bars),
                    "min_bars": self.min_bars_for_chart}
        return {
            "status": "OK",
            "bars": len(bars),
            "from_ts": bars[0].bucket_start,
            "to_ts": bars[-1].bucket_start,
            "bucket_seconds": self.bucket_seconds,
            "label": "observed candles (GROK6-collected)",
            "is_exchange_ohlcv": False,
        }

"""Layer A: token safety. Semantics locked down.

Rules:
- rugged != honeypot. Separate fields, separate claims.
- Missing critical data is NOT a pass.
- Hard reject only on positive confirmed danger signals:
    * mint authority PRESENT
    * freeze authority PRESENT
    * RugCheck rugged == True
- Missing authority data => NEEDS_REFRESH (retry), not reject.
- No sell route => NO_SELL_ROUTE rejection (not called a honeypot).
"""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import List, Optional

from config import Config
from providers import AuthorityStatus
from providers_service import EnrichedSnapshot


@dataclass
class SafetyOutcome:
    decision: str                       # PASS | HARD_REJECT | NEEDS_REFRESH | WAITING
    score: float
    reason_codes: List[str] = field(default_factory=list)
    evidence: dict = field(default_factory=dict)
    missing_fields: List[str] = field(default_factory=list)


class SafetyEvaluator:
    def __init__(self, cfg: Config):
        self.cfg = cfg

    def evaluate(self, snap: EnrichedSnapshot) -> SafetyOutcome:
        reasons: List[str] = []
        evidence: dict = {}
        missing: List[str] = []
        hard = False

        # ---- Positive confirmed danger: hard reject ----
        if snap.mint_authority == AuthorityStatus.PRESENT:
            hard = True
            reasons.append("MINT_AUTHORITY_PRESENT")
            evidence["mint_authority"] = "PRESENT"
        elif snap.mint_authority == AuthorityStatus.DISABLED:
            evidence["mint_authority"] = "DISABLED"
        else:
            missing.append("mint_authority")

        if snap.freeze_authority == AuthorityStatus.PRESENT:
            hard = True
            reasons.append("FREEZE_AUTHORITY_PRESENT")
            evidence["freeze_authority"] = "PRESENT"
        elif snap.freeze_authority == AuthorityStatus.DISABLED:
            evidence["freeze_authority"] = "DISABLED"
        else:
            missing.append("freeze_authority")

        if snap.rugcheck is not None and snap.rugcheck.status == "OK":
            if snap.rugcheck.rugged is True:
                hard = True
                reasons.append("RUGCHECK_RUGGED_TRUE")
                evidence["rugged"] = True
            elif snap.rugcheck.rugged is False:
                evidence["rugged"] = False
            else:
                missing.append("rugged")
            if snap.rugcheck.score_normalised is not None:
                evidence["rugcheck_score"] = snap.rugcheck.score_normalised
            if snap.rugcheck.top10_pct is not None:
                evidence["top10_pct"] = snap.rugcheck.top10_pct
                if snap.rugcheck.top10_pct > self.cfg.max_top_holder_pct:
                    reasons.append("TOP_HOLDERS_TOO_CONCENTRATED")
        else:
            missing.append("rugcheck")

        # Liquidity: missing -> NEEDS_REFRESH, low but present -> hard reject
        if snap.liquidity_usd is None:
            missing.append("liquidity_usd")
        elif snap.liquidity_usd < self.cfg.min_liquidity_usd:
            hard = True
            reasons.append("LIQUIDITY_TOO_LOW")
            evidence["liquidity_usd"] = snap.liquidity_usd
        else:
            evidence["liquidity_usd"] = snap.liquidity_usd

        # Sell route is advisory-but-important. Missing route -> REJECT, not honeypot.
        if snap.sell_route is not None:
            if snap.sell_route.status == "NO_ROUTE":
                hard = True
                reasons.append("NO_SELL_ROUTE")
            elif snap.sell_route.status == "SELL_ROUTE_UNKNOWN":
                missing.append("sell_route")
            else:
                evidence["sell_route_ok"] = True
                if snap.sell_route.price_impact_pct is not None:
                    evidence["sell_price_impact_pct"] = snap.sell_route.price_impact_pct
        else:
            missing.append("sell_route")

        # ----- Decision -----
        if hard:
            return SafetyOutcome("HARD_REJECT", 0.0, reasons, evidence, missing)

        # Missing critical-but-fetchable -> NEEDS_REFRESH
        critical_missing = [m for m in missing if m in
                            {"mint_authority", "freeze_authority", "rugcheck",
                             "liquidity_usd"}]
        if critical_missing:
            return SafetyOutcome("NEEDS_REFRESH", 0.3, reasons, evidence, missing)

        # Sell-route unknown only -> advisory but not fully clear
        if "sell_route" in missing:
            return SafetyOutcome("WAITING", 0.6, reasons + ["SELL_ROUTE_PENDING"],
                                 evidence, missing)

        # Score from evidence we have
        score = 1.0
        if "TOP_HOLDERS_TOO_CONCENTRATED" in reasons:
            score -= 0.3
        return SafetyOutcome("PASS", max(0.0, min(1.0, score)), reasons, evidence, missing)

    score: float
    reasons: List[str] = field(default_factory=list)
    safety_evidence: dict = field(default_factory=dict)
    missing_fields: List[str] = field(default_factory=list)
    agents: List[dict] = field(default_factory=list)
    risk: Optional[dict] = None
    ts: float = field(default_factory=time.time)


class Intelligence:
    def __init__(self, cfg: Config, storage: Storage,
                 providers_service: ProvidersService,
                 risk: RiskEngine, position_manager: PositionManager,
                 registry: Optional[CandidateRegistry] = None):
        self.cfg = cfg
        self.storage = storage
        self.providers = providers_service
        self.risk = risk
        self.pm = position_manager
        self.registry = registry or CandidateRegistry(default_ttl_seconds=3600)
        self.strategy = StrategyEngine(cfg)
        self.safety = SafetyEvaluator(cfg)
        self.agents = AgentTeam(cfg)
        self._last_scan_ts: Optional[float] = None
        self._last_decisions: List[Decision] = []

    def last_scan_ts(self) -> Optional[float]:
        return self._last_scan_ts

    def last_decisions(self) -> List[Decision]:
        return list(self._last_decisions)

    # ---------------- discovery callbacks ----------------
    def register_discovery(self, mint: str, source: str,
                           source_event_at: Optional[float] = None) -> Candidate:
        c = self.registry.get_or_create(mint, source, source_event_at)
        self.storage.upsert_candidate(mint, None, None,
                                      {"source": source,
                                       "source_event_at": source_event_at})
        self.storage.log("INFO", "intelligence", "DISCOVERED",
                         f"{mint} via {source}")
        return c

    # ---------------- evaluate ----------------
    def evaluate_candidate(self, mint: str, day: str) -> Decision:
        cand = self.registry.get(mint)
        if cand is None:
            cand = self.registry.get_or_create(mint, "unknown")

        if cand.is_expired():
            cand.transition(Stage.EXPIRED, "TTL_EXCEEDED")
            return self._decision_from_candidate(cand)

        snap = self.providers.enrich(mint)
        cand.symbol = snap.symbol
        cand.pool_id = snap.pool_id

        # --- market enrichment gate ---
        if snap.price_usd is None:
            if snap.dex_status == "NO_PAIR_YET":
                cand.transition(Stage.NEEDS_MARKET_ENRICHMENT, "NO_PAIR_YET",
                                "no DEX pair yet")
                cand.schedule_retry(delay_seconds=20, reason_code="NO_PAIR_YET")
                return self._decision_from_candidate(cand)
            cand.transition(Stage.STALE_DATA, "NO_PRICE",
                            "DexScreener returned no usable price")
            return self._decision_from_candidate(cand)

        # --- on-chain gate ---
        cand.transition(Stage.NEEDS_ONCHAIN_CHECK, "RPC_QUERIED")
        # --- risk / safety gate ---
        safety = self.safety.evaluate(snap)
        if safety.decision == "HARD_REJECT":
            cand.transition(Stage.REJECTED, "SAFETY_HARD_REJECT",
                            ",".join(safety.reason_codes))
            return self._decision_from_candidate(cand, safety=safety, snap=snap)
        if safety.decision == "NEEDS_REFRESH":
            cand.transition(Stage.NEEDS_REFRESH, "MISSING_SAFETY_DATA",
                            ",".join(safety.missing_fields))
            cand.schedule_retry(delay_seconds=45, reason_code="SAFETY")
            return self._decision_from_candidate(cand, safety=safety, snap=snap)
        if safety.decision == "WAITING":
            cand.transition(Stage.NEEDS_RISK_CHECK, "SELL_ROUTE_PENDING")
            cand.schedule_retry(delay_seconds=30, reason_code="SELL_ROUTE")
            return self._decision_from_candidate(cand, safety=safety, snap=snap)

        # --- strategy + agents ---
        plan = self.strategy.plan_entry_from_snapshot(snap)
        momentum = self.strategy.momentum_score_from_snapshot(snap)
        cand.transition(Stage.READY_FOR_DECISION, "GATES_PASSED")

        fast_launch = (
            self.cfg.fast_launch_enabled
            and not plan.rejected
            and plan.setup == Setup.EARLY_MOMENTUM
            and momentum["score"] >= 0.55
        )
        agents = self.agents.run_snapshot(snap, plan, safety, momentum,
                                          fast_launch=fast_launch)
        for a in agents:
            self.storage.record_agent_decision(mint, a.agent, a.status, a.score,
                                               a.reason_codes, a.evidence)

        if plan.rejected:
            code = plan.reject_reason or "STRATEGY_REJECT"
            cand.transition(Stage.REJECTED, code, ",".join(plan.reasons))
            return self._decision_from_candidate(cand, safety=safety, snap=snap,
                                                 plan=plan, momentum=momentum,
                                                 agents=agents)

        # --- risk ---
        risk = self.risk.evaluate(day, plan.entry_price, plan.stop_price,
                                  category="fast_launch" if fast_launch else "technical")
        if not risk.approved:
            cand.transition(Stage.REJECTED, f"RISK_{risk.reason}", risk.reason)
            return self._decision_from_candidate(cand, safety=safety, snap=snap,
                                                 plan=plan, momentum=momentum,
                                                 agents=agents, risk=risk)

        cand.transition(Stage.ELIGIBLE_PAPER, "ALL_GATES_PASSED")

        # --- paper open ---
        pos = self.pm.open_position(
            mint=mint, symbol=snap.symbol, setup=plan.setup.value,
            category="fast_launch" if fast_launch else "technical",
            entry_ref_price=plan.entry_price, stop_price=plan.stop_price,
            tp1_price=plan.tp1_price, tp2_price=plan.tp2_price,
            qty=risk.qty, entry_reason=",".join(plan.reasons),
            safety=safety.evidence,
        )
        if pos:
            cand.transition(Stage.PAPER_OPEN, "PAPER_OPENED")
        else:
            cand.transition(Stage.REJECTED, "PAPER_OPEN_FAILED")

        return self._decision_from_candidate(cand, safety=safety, snap=snap,
                                             plan=plan, momentum=momentum,
                                             agents=agents, risk=risk)

    def scan_once(self, day: str) -> List[Decision]:
        self.registry.expire_due()
        mints = [c.mint for c in self.registry.all()
                 if c.stage not in TERMINAL]
        decisions: List[Decision] = []
        for mint in mints:
            try:
                decisions.append(self.evaluate_candidate(mint, day))
            except Exception as e:
                log.exception("evaluate failed for %s", mint)
                self.storage.log("ERROR", "intelligence", "EVAL_ERROR",
                                 f"{mint}: {type(e).__name__}: {e}")
        self._last_scan_ts = time.time()
        self._last_decisions = decisions
        return decisions

    def manage_open_positions(self) -> None:
        opens = self.storage.list_positions(statuses=["OPEN", "TP1_HIT"])
        for p in opens:
            snap = self.providers.enrich(p["mint"])
            self.pm.manage(p["id"], snap.price_usd)

    # ---------------- helpers ----------------
    def _decision_from_candidate(self, c: Candidate, safety=None, snap=None,
                                 plan=None, momentum=None, agents=None,
                                 risk=None) -> Decision:
        d = Decision(
            mint=c.mint, symbol=c.symbol, stage=c.stage.value,
            reason_code=c.reason_code or "UNKNOWN",
            block_reason=c.block_reason or "",
            safety_decision=getattr(safety, "decision", "UNKNOWN") if safety else "UNKNOWN",
            setup=getattr(plan, "setup", Setup.NONE).value if plan else Setup.NONE.value,
            score=(momentum or {}).get("score", 0.0) if momentum else 0.0,
            reasons=list(getattr(plan, "reasons", []) if plan else []),
            safety_evidence=getattr(safety, "evidence", {}) if safety else {},
            missing_fields=getattr(safety, "missing_fields", []) if safety else [],
            agents=[a.__dict__ for a in agents] if agents else [],
            risk=risk.__dict__ if risk else None,
        )
        payload = {
            "stage": d.stage, "reason_code": d.reason_code,
            "block_reason": d.block_reason,
            "safety_decision": d.safety_decision,
            "setup": d.setup, "score": d.score,
            "reasons": d.reasons,
            "safety_evidence": d.safety_evidence,
            "missing_fields": d.missing_fields,
        }
        self.storage.record_candidate_history(c.mint, d.stage, d.score,
                                              d.reason_code, payload)
        return d

    # ---------------- health ----------------
    def health(self) -> Dict[str, Any]:
        now = time.time()
        last = self._last_scan_ts
        age = None if last is None else now - last
        if last is None:
            state = "UNHEALTHY"
        elif age and age > self.cfg.scan_interval_seconds * 5:
            state = "UNHEALTHY"
        elif age and age > self.cfg.scan_interval_seconds * 2:
            state = "DEGRADED"
        else:
            state = "HEALTHY"
        return {
            "state": state,
            "last_scan_ts": last,
            "last_scan_age_seconds": age,
            "candidates_total": len(self.registry.all()),
            "candidates_active": sum(1 for c in self.registry.all()
                                     if c.stage not in TERMINAL),
        }

def momentum_score_from_snapshot(self, snap) -> dict:
    contributions = {}
    score = 0.0
    v5 = snap.volume_5m_usd
    v1h = snap.volume_1h_usd
    if v5 is not None and v1h and v1h > 0:
        ratio = (v5 * 12) / v1h
        c = min(max((ratio - 1.0) * 0.3, 0.0), 0.4)
        contributions["volume_acceleration"] = c
        score += c
    b, s = snap.buys_5m, snap.sells_5m
    if b is not None and s is not None and (b + s) > 0:
        imb = (b - s) / (b + s)
        c = max(0.0, imb) * 0.25
        contributions["buy_sell_imbalance"] = c
        score += c
    pc = snap.price_change_5m_pct
    if pc is not None:
        if 1.0 <= pc <= 30.0:
            c = min((pc / 30.0) * 0.25, 0.25)
            contributions["price_change_5m"] = c
            score += c
        elif pc > 40.0:
            contributions["price_change_5m"] = -0.1
            score -= 0.1
    liq = snap.liquidity_usd
    if liq is not None and liq >= self.cfg.min_liquidity_usd:
        import math
        c = min(math.log10(liq / self.cfg.min_liquidity_usd + 1) * 0.1, 0.1)
        contributions["liquidity_depth"] = c
        score += c
    return {"score": max(0.0, min(score, 1.0)), "contributions": contributions}

    # callbacks from pump ws
    async def _on_new_token(self, msg: dict) -> None:
        mint = msg.get("mint") or msg.get("tokenAddress")
        if not mint:
            return
        ts = msg.get("timestamp")
        if isinstance(ts, (int, float)) and ts > 1e12:
            ts /= 1000.0
        self.intelligence.register_discovery(mint, "pumpportal:newToken", ts)

    async def _on_migration(self, msg: dict) -> None:
        mint = msg.get("mint") or msg.get("tokenAddress")
        if not mint:
            return
        self.intelligence.register_discovery(mint, "pumpportal:migration")

    # pump ws runs in its own thread with its own event loop
    def _pump_worker(self) -> None:
        loop = asyncio.new_event_loop()
        self._pump_loop = loop
        asyncio.set_event_loop(loop)
        try:
            loop.run_until_complete(self.pumpportal.run())
        finally:
            loop.close()

    def _start_pump_ws(self) -> None:
        if not self.pumpportal:
            return
        self._pump_thread = threading.Thread(target=self._pump_worker,
                                             name="grok6-pumpportal", daemon=True)
        self._pump_thread.start()
        self.storage.log("INFO", "app", "PUMPPORTAL_THREAD_STARTED", "")

    # a small tick inside discovery loop records candles
    def _record_candles(self) -> None:
        for c in self.intelligence.registry.all():
            snap = self.providers.enrich(c.mint)
            if snap.price_usd:
                self.candles.add_sample(c.mint, snap.price_usd)

    def today(self) -> str:
        return dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%d")

    def run_server(self) -> None:
        app_state = {
            "cfg": self.cfg,
            "storage": self.storage,
            "providers": self.providers,
            "intelligence": self.intelligence,
            "discovery": self.discovery,
            "risk": self.risk,
            "pm": self.pm,
            "candles": self.candles,
            "today": self.today,
            "pumpportal": self.pumpportal,
        }
        app = build_dashboard(app_state)
        self._start_pump_ws()
        self.discovery.start()
        self.storage.log("INFO", "app", "SERVER_START",
                         f"http://{self.cfg.host}:{self.cfg.port}")
        try:
            app.run(host=self.cfg.host, port=self.cfg.port,
                    debug=False, use_reloader=False)
        finally:
            self.discovery.stop()
            if self.pumpportal and self._pump_loop:
                fut = asyncio.run_coroutine_threadsafe(self.pumpportal.stop(),
                                                       self._pump_loop)
                try:
                    fut.result(timeout=5)
                except Exception:
                    pass
            self.storage.close()
